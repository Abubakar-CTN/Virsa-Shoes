function getStock(product) {
    if (
        product.stock !== undefined &&
        product.stock !== null &&
        product.stock !== "" &&
        Number.isFinite(Number(product.stock))
    ) {
        return Math.max(0, Number(product.stock));
    }

    if (Array.isArray(product.colorVariants)) {
        return product.colorVariants.reduce(
            (total, variant) =>
                total + Math.max(0, Number(variant.stock) || 0),
            0,
        );
    }

    return 0;
}

    const [notificationsOpen, setNotificationsOpen] = useState(false);
    const [readNotificationIds, setReadNotificationIds] = useState(
        () => new Set(),
    );
    const notificationRef = useRef(null);

    const notifications = useMemo(
        () =>
            catalogue
                .filter((product) => getStock(product) <= LOW_STOCK_LIMIT)
                .sort((a, b) => getStock(a) - getStock(b)),
        [catalogue],
    );

    const unreadCount = notifications.filter(
        (product) => !readNotificationIds.has(String(product.id)),
    ).length;

    useEffect(() => {
        function refreshCatalogue(event) {
            const latestProducts = event.detail?.products;

            setCatalogue(
                Array.isArray(latestProducts)
                    ? latestProducts
                    : getProducts(),
            );

            setReadNotificationIds(new Set());
        }

        window.addEventListener(
            "virsa:products-updated",
            refreshCatalogue,
        );

        return () => {
            window.removeEventListener(
                "virsa:products-updated",
                refreshCatalogue,
            );
        };
    }, []);

    useEffect(() => {
        if (!notificationsOpen) return undefined;

        function handleOutsideClick(event) {
            if (!notificationRef.current?.contains(event.target)) {
                setNotificationsOpen(false);
            }
        }

        function handleEscape(event) {
            if (event.key === "Escape") {
                setNotificationsOpen(false);
            }
        }

        document.addEventListener("mousedown", handleOutsideClick);
        document.addEventListener("keydown", handleEscape);

        return () => {
            document.removeEventListener("mousedown", handleOutsideClick);
            document.removeEventListener("keydown", handleEscape);
        };
    }, [notificationsOpen]);

    function markAllRead() {
        setReadNotificationIds(
            new Set(notifications.map((product) => String(product.id))),
        );

        setNotificationsOpen(false);

    const [products, setProducts] = useState(() => getProducts());
    const [notificationsOpen, setNotificationsOpen] = useState(false);
    const notificationRef = useRef(null);

    useEffect(() => {
        function syncProducts(event) {
            const updatedProducts = event.detail?.products;

            setProducts(
                Array.isArray(updatedProducts)
                    ? updatedProducts
                    : getProducts(),
            );
        }

        window.addEventListener("virsa:products-updated", syncProducts);

        return () => {
            window.removeEventListener(
                "virsa:products-updated",
                syncProducts,
            );
        };
    }, []);

    useEffect(() => {
        if (!notificationsOpen) return undefined;

        function handleOutsideClick(event) {
            if (!notificationRef.current?.contains(event.target)) {
                setNotificationsOpen(false);
            }
        }

        function handleEscape(event) {
            if (event.key === "Escape") {
                setNotificationsOpen(false);
            }
        }

        document.addEventListener("mousedown", handleOutsideClick);
        document.addEventListener("keydown", handleEscape);

        return () => {
            document.removeEventListener("mousedown", handleOutsideClick);
            document.removeEventListener("keydown", handleEscape);
        };
    }, [notificationsOpen]);

    const notifications = useMemo(() => {
        return products
            .map((product) => {
                const stock = getStock(product);

                if (stock === 0) {
                    return {
                        id: `out-${product.id}`,
                        type: "out",
                        title: "Out of stock",
                        name: product.name || "Unnamed product",
                        description: "No units currently available.",
                        priority: 0,
                    };
                }

                if (stock <= LOW_STOCK_LIMIT) {
                    return {
                        id: `low-${product.id}`,
                        type: "low",
                        title: "Low stock",
                        name: product.name || "Unnamed product",
                        description: `${stock} ${stock === 1 ? "unit" : "units"
                            } remaining. Review availability.`,
                        priority: 1,

                return null;

        },
        mutations: {
